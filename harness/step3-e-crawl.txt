# Step 3, crawl pose (Lie Flat Chance 0 %): a heel cut drops a fighting NPC into the game's own
# injured/crawl animations (standing ragdoll, low Height, Injured = 1). It stays alive
# and its AI keeps moving it, slowly, which the animator plays as crawling.
# Pictures over time; then, with Lie Flat Chance 100 %, a second NPC lies flat.
wait-level
set-option AchillesTendon "Achilles Tendon Enabled" true
set-option AchillesTendon "Instant Death Chance" 0
set-option AchillesTendon "Stand Up When Disabled" true
set-option AchillesTendon "Allowed Damage Types" SlashAndPierce
set-option AchillesTendon "Debug Logging" true
set-option AchillesTendon "Max Hit Angle" 110
set-option AchillesTendon "Lie Flat Chance" 0
set-option AchillesTendon "Kneel Chance" 0
set-option AchillesTendon "Crawl Speed" 0.3
set-option AchillesTendon "Crawl Body Height" 0.2
set-option AchillesTendon "Bleed-Out Enabled" false
set-option AchillesTendon "Ground Fighting Enabled" true
set-option AchillesTendon "Screaming" true
set-option AchillesTendon "Ground Attacks" true
set-option AchillesTendon "Ground Attack Chance" 1
set-option AchillesTendon "Ground Attack Range" 4
set-option AchillesTendon "Ground Attack Interval" 2

# A real fighter (HumanMedium brain) with a sword, back turned, 3 m away.
spawn-npc HumanMale distance=3 facing=180 brain=HumanMedium
npc-grab SwordShortCommon
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] HEEL HIT
expect-log [AchillesTendon] COLLAPSE
expect-log down to the injured/crawl pose
expect-npc state Alive within=1
wait 1
screenshot 01-crawl-dropping of=npc distance=2.5 height=1.0 angle=90
wait 2
screenshot 02-crawl-3s of=npc distance=2.5 height=1.0 angle=90
wait 2
screenshot 03-crawl-5s of=npc distance=2.5 height=1.0 angle=90
wait 2
screenshot 04-crawl-7s of=npc distance=2.5 height=1.0 angle=45
wait 2
screenshot 05-crawl-9s of=npc distance=2.5 height=1.0 angle=135
# Its own melee AI is off while crawling; swings come on the upper-body layer.
expect-log upper body | clip=
screenshot 05a-swing of=npc distance=2.5 height=1.0 angle=90
wait 0.25
screenshot 05b-swing of=npc distance=2.5 height=1.0 angle=90
wait 0.25
screenshot 05c-swing of=npc distance=2.5 height=1.0 angle=90
wait 0.25
screenshot 05d-swing of=npc distance=2.5 height=1.0 angle=90
expect-npc state Alive within=0.1
expect-no-log Exception
# The harness logs its own step lines; a new mark keeps "expect-no-log Exception" from matching them.
mark

# Lie Flat Chance 100 %: the next cut NPC is knocked down and held flat.
set-option AchillesTendon "Lie Flat Chance" 1
# A dummy: a fighter turned toward the player (already in combat) within the 0.12 s between spawn
# and cut, and the cut came from the front (run 20261124-150347: angleFromBehind=130.3, miss).
spawn-npc HumanMale distance=2 angle=60 facing=180 brain=HumanDummy
npc-grab SwordShortCommon
hit-npc LeftFoot type=Slash
expect-log held on the ground (roll
expect-npc state Destabilized within=2
wait 1
screenshot 06-lie-flat of=npc distance=2.5 height=1.0 angle=90
set-option AchillesTendon "Lie Flat Chance" 0
set-option AchillesTendon "Kneel Chance" 0

set-option AchillesTendon "Bleed-Out Enabled" true
expect-no-log Exception
quit
