# Heel of Achilles: loads headless, its script starts, every Mod Menu option is there
# with its default, options apply live, the level runs without exceptions.
# Run: tools/harness.sh harness/smoke.txt
# Checks the defaults, so it must be the first scenario of a run (fresh prefix);
# scenarios in one run share the prefix and the game keeps option values.
wait-level
expect-log [ModManager][ModOption][AchillesTendon] Loaded 30 ModOptions
expect-log [AchillesTendon] Loaded Heel of Achilles
expect-log [ModManager][ThunderScript][AchillesTendon] Loaded ThunderScript: AchillesTendon.AchillesTendonScript
expect-log [AchillesTendon] Enabled, listening to EventManager.onCreatureHit

# Achilles Tendon group (step 3 adds Stand Up When Disabled)
option-exists AchillesTendon "Achilles Tendon Enabled" category="Achilles Tendon" type=Boolean save=true
option-exists AchillesTendon "Stand Up When Disabled" category="Achilles Tendon" type=Boolean save=true tooltip="get up again"
option-exists AchillesTendon "Downed Pose" category="Achilles Tendon" type=DownedPose save=true
option-exists AchillesTendon "Crawl Speed" category="Achilles Tendon" type=Single save=true
option-exists AchillesTendon "Crawl Body Height" category="Achilles Tendon" type=Single save=true
# Bleed-Out group (step 5)
option-exists AchillesTendon "Bleed-Out Enabled" category="Bleed-Out" type=Boolean save=true
option-exists AchillesTendon "Bleed-Out Duration" category="Bleed-Out" type=Single save=true
option-exists AchillesTendon "Blood Effects" category="Bleed-Out" type=Boolean save=true
option-exists AchillesTendon "Blood Pulse Interval" category="Bleed-Out" type=Single save=true
option-exists AchillesTendon "Weakening" category="Bleed-Out" type=Boolean save=true
option-exists AchillesTendon "Weakest Crawl Speed" category="Bleed-Out" type=Single save=true
option-exists AchillesTendon "Scream Slowdown" category="Bleed-Out" type=Single save=true
option-exists AchillesTendon "Gasping From" category="Bleed-Out" type=Single save=true
# Ground Fighting group (step 4, experimental)
option-exists AchillesTendon "Ground Fighting Enabled" category="Ground Fighting" type=Boolean save=true
option-exists AchillesTendon "Arm Strength" category="Ground Fighting" type=Single save=true
option-exists AchillesTendon "Ground Attacks" category="Ground Fighting" type=Boolean save=true
option-exists AchillesTendon "Ground Attack Range" category="Ground Fighting" type=Single save=true
option-exists AchillesTendon "Ground Attack Interval" category="Ground Fighting" type=Single save=true
option-exists AchillesTendon "Screaming" category="Ground Fighting" type=Boolean save=true
option-exists AchillesTendon "Scream Interval" category="Ground Fighting" type=Single save=true
option-exists AchillesTendon "Apply To Player" category="Player Hardcore Mode" type=Boolean save=true
option-exists AchillesTendon "Debug Logging" category=Debug type=Boolean save=true

# Defaults
expect-option AchillesTendon "Achilles Tendon Enabled" true
expect-option AchillesTendon "Stand Up When Disabled" true
expect-option AchillesTendon "Downed Pose" Crawl
expect-option AchillesTendon "Crawl Speed" 0.3
expect-option AchillesTendon "Crawl Body Height" 0.2
expect-option AchillesTendon "Bleed-Out Enabled" true
expect-option AchillesTendon "Bleed-Out Duration" 45
expect-option AchillesTendon "Blood Effects" true
expect-option AchillesTendon "Blood Pulse Interval" 2
expect-option AchillesTendon "Weakening" true
expect-option AchillesTendon "Weakest Crawl Speed" 0.25
expect-option AchillesTendon "Scream Slowdown" 3
expect-option AchillesTendon "Gasping From" 0.75
expect-option AchillesTendon "Ground Fighting Enabled" true
expect-option AchillesTendon "Arm Strength" 1
expect-option AchillesTendon "Ground Attacks" false
expect-option AchillesTendon "Ground Attack Range" 1.8
expect-option AchillesTendon "Ground Attack Interval" 3
expect-option AchillesTendon "Screaming" true
expect-option AchillesTendon "Scream Interval" 6
expect-option AchillesTendon "Apply To Player" false
expect-option AchillesTendon "Debug Logging" false

# Live toggles reach the static fields
set-option AchillesTendon "Stand Up When Disabled" false
expect-option AchillesTendon "Stand Up When Disabled" false
set-option AchillesTendon "Achilles Tendon Enabled" false
expect-option AchillesTendon "Achilles Tendon Enabled" false
set-option AchillesTendon "Achilles Tendon Enabled" true
set-option AchillesTendon "Stand Up When Disabled" true
wait 3

expect-no-log Exception
screenshot achilles-arena
quit
