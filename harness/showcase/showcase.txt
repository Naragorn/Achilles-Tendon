# Showcase: every step of the mod in pictures, in the order a player meets them.
# Not part of the default run (subfolder). NPCs are not despawned (no harness verb), so each
# new one stands at another angle around the player. Run: tools/harness.sh harness/showcase/showcase.txt
# Pictures land in artifacts/harness/<run>/showcase/. Debug views (heel zones, hit markers)
# are on so the pictures show what the mod checks.
wait-level
set-option AchillesTendon "Achilles Tendon Enabled" true
set-option AchillesTendon "Stand Up When Disabled" true
set-option AchillesTendon "Allowed Damage Types" SlashAndPierce
set-option AchillesTendon "Max Hit Angle" 110
set-option AchillesTendon "Downed Pose" Crawl
set-option AchillesTendon "Debug Logging" true
set-option AchillesTendon "Show Heel Zones" true
set-option AchillesTendon "Show Hit Markers" true
set-option AchillesTendon "Hit Marker Duration" 10
set-option AchillesTendon "Bleed-Out Enabled" true
set-option AchillesTendon "Bleed-Out Duration" 30
set-option AchillesTendon "Blood Effects" true
set-option AchillesTendon "Blood Pulse Interval" 1
set-option AchillesTendon "Weakening" true
set-option AchillesTendon "Ground Fighting Enabled" true
set-option AchillesTendon "Screaming" true
set-option AchillesTendon "Ground Attacks" true
set-option AchillesTendon "Ground Attack Range" 4
set-option AchillesTendon "Ground Attack Interval" 2

# 1. Refused: a blunt hit on the heel. Red marker, the NPC stays on its feet.
spawn-npc HumanMale distance=1.5 angle=-60 facing=180 brain=HumanDummy
wait 1
screenshot s01-heel-zones of=npc distance=1.3 height=0.25 angle=25
mark
hit-npc LeftFoot type=Blunt
expect-log miss (DamageTypeNotAllowed)
wait 0.5
screenshot s02-blunt-refused of=npc distance=1.3 height=0.25 angle=25
wait 2
expect-npc state Alive within=0.1
screenshot s03-still-standing of=npc distance=2.5 height=1.0 angle=90

# 2. Refused: a slash from the front.
spawn-npc HumanMale distance=1.5 angle=-30 facing=0 brain=HumanDummy
wait 1
mark
hit-npc LeftFoot type=Slash
expect-log miss (AttackAngleTooFrontal)
wait 0.5
screenshot s04-frontal-refused of=npc distance=1.3 height=0.25 angle=25

# 3. The cut: a fighter with a sword, back turned. Slash on the heel from behind.
spawn-npc HumanMale distance=3 facing=180 brain=HumanMedium
npc-grab SwordShortCommon
wait 0.5
screenshot s05-fighter of=npc distance=2.5 height=1.0 angle=30
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] HEEL HIT
expect-log [AchillesTendon] COLLAPSE
screenshot s06-cut of=npc distance=1.3 height=0.25 angle=25
wait 0.5
screenshot s07-falling of=npc distance=2.5 height=1.0 angle=90
wait 1.5
screenshot s08-down of=npc distance=2.5 height=1.0 angle=90

# 4. Crawling towards the player, swinging from the ground.
wait 3
screenshot s09-crawl-5s of=npc distance=2.5 height=1.0 angle=90
wait 3
screenshot s10-crawl-8s of=npc distance=2.5 height=1.0 angle=45
expect-log upper body | clip=
screenshot s11-swing-a of=npc distance=2.5 height=1.0 angle=90
wait 0.25
screenshot s12-swing-b of=npc distance=2.5 height=1.0 angle=90
wait 0.25
screenshot s13-swing-c of=npc distance=2.5 height=1.0 angle=90

# 5. Bleeding: drip at the cut heel, weaker over time.
wait 2
screenshot s14-blood of=npc distance=1.0 height=0.3 angle=150
wait 8
screenshot s15-bleeding-20s of=npc distance=2.5 height=1.0 angle=90
wait 5
screenshot s16-near-death of=npc distance=2.5 height=1.0 angle=90

# 6. Bled out.
expect-npc state Dead within=15
expect-log [AchillesTendon] BLED OUT
wait 1
screenshot s17-bled-out of=npc distance=2.5 height=1.0 angle=90

# 7. The other pose: Lie Flat.
set-option AchillesTendon "Downed Pose" LieFlat
set-option AchillesTendon "Bleed-Out Enabled" false
spawn-npc HumanMale distance=1.5 angle=60 facing=180 brain=HumanDummy
npc-grab SwordShortCommon
wait 0.5
mark
hit-npc LeftFoot type=Slash
expect-npc state Destabilized within=2
wait 2
screenshot s18-lie-flat of=npc distance=2.5 height=1.0 angle=90

# 8. Finished off by the player.
hit-npc Head type=Slash damage=999
expect-npc state Dead within=3
wait 1
screenshot s19-finished of=npc distance=2.5 height=1.0 angle=90

set-option AchillesTendon "Downed Pose" Crawl
set-option AchillesTendon "Bleed-Out Enabled" true
set-option AchillesTendon "Show Heel Zones" false
set-option AchillesTendon "Show Hit Markers" false
expect-no-log Exception
quit
