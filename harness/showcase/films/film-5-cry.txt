# Film 5: the collapse cry (v0.8.9), with sound. BasHarness films have no audio track:
#   tools/film-with-sound.sh harness/showcase/films/film-5-cry.txt
# records the game's own audio on a sink of its own and lays it under each film (<name>-sound.mp4).
# A man and a woman (each NPC gets a random voice), each cut from behind: the cry at the cut, then
# quiet until the first pain scream one Scream Interval (6 s) later. Runs on v0.8.8 too (the "before"
# film: the plain pain scream right at the cut), so it checks only what both versions log.
wait-level
# Headset pace (see film-1-sit.txt).
frame-rate 90
set-option AchillesTendon "Achilles Tendon Enabled" true
set-option AchillesTendon "Instant Death Chance" 0
set-option AchillesTendon "Max Hit Angle" 110
set-option AchillesTendon "Debug Logging" true
set-option AchillesTendon "Show Heel Zones" false
set-option AchillesTendon "Show Hit Markers" false
set-option AchillesTendon "Lie Flat Chance" 0
set-option AchillesTendon "Kneel Chance" 0
set-option AchillesTendon "Bleed-Out Enabled" true
set-option AchillesTendon "Ground Fighting Enabled" true
set-option AchillesTendon "Screaming" true
set-option AchillesTendon "Scream Interval" 6
player-invincible true

spawn-npc HumanMale distance=2.5 facing=180 brain=HumanDummy
wait 1.5
record-start cry-male fps=30 width=960 height=540 views=npc distance=2.6 rise=1.0 angle=90
wait 1
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] COLLAPSE |
wait 7.5
record-stop
expect-log [AchillesTendon] SCREAM
# Killed, so its screams stay out of the next film.
hit-npc Head type=Pierce damage=500
expect-npc state Dead within=2
wait 2

spawn-npc HumanFemale distance=2.5 angle=-60 facing=180 brain=HumanDummy
wait 1.5
record-start cry-female fps=30 width=960 height=540 views=npc distance=2.6 rise=1.0 angle=90
wait 1
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] COLLAPSE |
wait 7.5
record-stop
expect-log [AchillesTendon] SCREAM
expect-no-log Exception
quit
