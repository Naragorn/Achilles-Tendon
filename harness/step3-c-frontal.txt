# Step 3: a slash on the heel while the NPC faces the player is too frontal
# (default Max Hit Angle 110), so the NPC stays up.
wait-level
# Scenarios in one run share the harness prefix, and the game keeps option values
# across them, so every scenario sets the options it depends on.
set-option AchillesTendon "Achilles Tendon Enabled" true
set-option AchillesTendon "Stand Up When Disabled" true
set-option AchillesTendon "Allowed Damage Types" SlashAndPierce
set-option AchillesTendon "Debug Logging" true
set-option AchillesTendon "Max Hit Angle" 110
spawn-npc HumanMale distance=1.5 facing=0 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] miss (AttackAngleTooFrontal)
wait 3
expect-npc state Alive within=0.1
expect-no-log Exception
quit
