# DMCS compat: a heel cut and the player's kill of a crippled NPC fill DMCS's Adrenaline meter
# (DMCS.Adrenaline.AddExternal). Needs DMCS loaded too, so it is not in the default run:
#   DMCS_MOD=../DMCS/src/DMCS/bin/Release/net472 tools/harness.sh harness/compat/dmcs-adrenaline.txt
# Each NPC spawns on its own spot (3.5 m from the player at -120..120 degrees, two at 1.5 m; the next case
# at least 4 m away), never on an earlier body: in run 20261118-223801 case 8's dummy spawned 0.27 s after
# case 7's kill on the same spot, took 1.6 damage before its cut, stood 12 degrees turned, and the heel hit
# missed the zone (0.082 m, radius 0.080 m). Not behind the player: at 3.4 m and -160 degrees the Arena
# ground is 1.26 m high, and a dummy spawned there died inside it (run 20261120-111650).
wait-level
set-option AchillesTendon "Achilles Tendon Enabled" true
set-option AchillesTendon "Instant Death Chance" 0
set-option AchillesTendon "Debug Logging" true
set-option AchillesTendon "Max Hit Angle" 110
set-option AchillesTendon "Lie Flat Chance" 0
set-option AchillesTendon "Kneel Chance" 0
set-option AchillesTendon "Bleed-Out Enabled" false
set-option AchillesTendon "DMCS Adrenaline" true
set-option AchillesTendon "Heel Cut Adrenaline" 20
set-option AchillesTendon "Crippled Kill Adrenaline" 35
set-option DMCS "Adrenaline Enabled" true
set-option DMCS "Debug Logging" true
# A full meter would ignore further gains (and its critical would kill the test NPCs): DMCS's own
# hit gain off and the largest meter keep it filling. Sum here: 20 + 35 + 20 + 20 = 95 of 300.
set-option DMCS "Adrenaline Per Hit" 0
set-option DMCS "Adrenaline Meter Max" 300

# 1. Heel cut: +20; then the player's sword kills the crawling NPC: +35.
spawn-npc HumanMale distance=3.5 angle=0 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] COLLAPSE
expect-log [AchillesTendon] DMCS | AchillesHeelCut +20
expect-log | Applied
wait 2
mark
hit-npc Head type=Slash damage=999 by=held item=SwordShortCommon
expect-npc state Dead within=3
expect-log [AchillesTendon] DMCS | AchillesCrippledKill +35
expect-log | Applied

# 2. A heel cut that kills at once: the cut counts, the kill is no execution.
spawn-npc HumanMale distance=3.5 angle=120 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash damage=999
expect-npc state Dead within=3
expect-log [AchillesTendon] DMCS | AchillesHeelCut +20
wait 1
expect-no-log AchillesCrippledKill

# 3. Bleeding out is no execution.
set-option AchillesTendon "Bleed-Out Enabled" true
set-option AchillesTendon "Bleed-Out Duration" 10
spawn-npc HumanMale distance=3.5 angle=-120 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] DMCS | AchillesHeelCut +20
expect-npc state Dead within=15
expect-log [AchillesTendon] BLED OUT
wait 1
expect-no-log AchillesCrippledKill
set-option AchillesTendon "Bleed-Out Enabled" false

# 4. A kill of an NPC that was never crippled gives nothing.
spawn-npc HumanMale distance=3.5 angle=40 facing=0 brain=HumanDummy
mark
hit-npc Head type=Slash damage=999 by=held item=SwordShortCommon
expect-npc state Dead within=3
wait 1
expect-no-log [AchillesTendon] DMCS |

# 5. Compat off: neither the cut nor the kill reaches DMCS.
set-option AchillesTendon "DMCS Adrenaline" false
spawn-npc HumanMale distance=1.5 angle=-60 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] COLLAPSE
wait 2
hit-npc Head type=Slash damage=999 by=held item=SwordShortCommon
expect-npc state Dead within=3
wait 1
expect-no-log [AchillesTendon] DMCS |
set-option AchillesTendon "DMCS Adrenaline" true

# 6-9. Every way the player kills counts as an execution: fist, kick, thrown blade, spell.
# GameAdapters.Attacker resolves fist and kick through the hitting ragdoll part (not
# CollisionInstance.IsDoneByPlayer). Smaller amounts keep the meter from filling:
# 4 x (5 + 10) = 60, so 155 of 300 after case 9 (160 after case 10).
set-option AchillesTendon "Heel Cut Adrenaline" 5
set-option AchillesTendon "Crippled Kill Adrenaline" 10

# 6. Fist
spawn-npc HumanMale distance=3.5 angle=80 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] DMCS | AchillesHeelCut +5
wait 2
mark
hit-npc Head type=Blunt damage=999 by=hand
expect-npc state Dead within=3
expect-log [AchillesTendon] DMCS | AchillesCrippledKill +10
expect-log | Applied

# 7. Kick (the player's right foot part)
spawn-npc HumanMale distance=3.5 angle=-80 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] DMCS | AchillesHeelCut +5
wait 2
mark
hit-npc Head type=Blunt damage=999 by=foot
expect-npc state Dead within=3
expect-log [AchillesTendon] DMCS | AchillesCrippledKill +10
expect-log | Applied

# 8. Thrown dagger (nobody holds it, the player was the last to)
spawn-npc HumanMale distance=1.5 angle=60 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] DMCS | AchillesHeelCut +5
wait 2
mark
hit-npc Head type=Pierce damage=999 by=thrown item=DaggerCommon
expect-npc state Dead within=3
expect-log [AchillesTendon] DMCS | AchillesCrippledKill +10
expect-log | Applied

# 9. Spell (casterHand)
spawn-npc HumanMale distance=3.5 angle=-40 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash
expect-log [AchillesTendon] DMCS | AchillesHeelCut +5
wait 2
mark
hit-npc Head type=Energy damage=999 by=spell
expect-npc state Dead within=3
expect-log [AchillesTendon] DMCS | AchillesCrippledKill +10
expect-log | Applied

# 10. Instant Death: the cut kills at once and counts as a heel cut (+5); the kill is the cutting hit on an
# NPC that was never crippled, so no execution (as in case 2). DMCS's own hit handlers then see a dead NPC
# at the hit's end (Creature.Damage raises OnEnd after a kill inside OnStart): no exception.
# 140 degrees, 2.5 m: 6 m from case 9, at least 1.4 m from every earlier body (case 2 at 120 degrees, 3.5 m).
set-option AchillesTendon "Instant Death Chance" 1
spawn-npc HumanMale distance=2.5 angle=140 facing=180 brain=HumanDummy
mark
hit-npc LeftFoot type=Slash by=held item=SwordShortCommon
expect-log [AchillesTendon] INSTANT DEATH
expect-npc state Dead within=1
expect-log [AchillesTendon] DMCS | AchillesHeelCut +5
expect-log | Applied
wait 1
expect-no-log AchillesCrippledKill
expect-no-log Adrenaline error
set-option AchillesTendon "Instant Death Chance" 0

expect-no-log Exception
quit
